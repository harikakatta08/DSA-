import java.util.*;
class maxSubArraySum {
    public static void main(String[] args) {
        int arr[]={1,2,3,4,5,6};
        int k=3;
        int max=0;
        int wsum=0;
        for(int i=0;i<k;i++){
            wsum+=arr[i];
        }
        for(int i=k;i<arr.length;i++){
            wsum+=arr[i];
            wsum-=arr[i-k];
            if(wsum>max){
                max=wsum;
            }
        }
        System.out.println(max + " ");
    }
}
