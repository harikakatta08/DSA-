class Prime {
    public static void main(String[] args) {
        int n=2;
        boolean isPrime=true;
        if(n<=1){
            isPrime=false;
        }
        for(int i=2;i<Math.sqrt(n);i++){
            if(n%i==0){
                isPrime= false;
            }
            
        }
        if(isPrime){
            System.out.println("Prime number");
            
        }
        else{
            System.out.println("Not a Prime");
        }
    }
}
