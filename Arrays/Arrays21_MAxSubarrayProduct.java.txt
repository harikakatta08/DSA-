class Maxsubarrayproduct{
    public static void main(String[] args) {
        int arr[]={1,2,3,4,5,6};
        int k=3;
        int wprod=1;
        int max=0;
        for(int i=0;i<k;i++){
            wprod*=arr[i];
        }
        for(int i=k;i<arr.length;i++){
           wprod = (wprod / arr[i-k]) * arr[i];
            if(wprod>max){
                max=wprod;
            }
            
        }
        System.out.println(max);
    }
}
