class PrimeInRange {
    public static boolean prime(int n){
        boolean isPrime=true;
        if(n<=1) return false;
        for(int i=2;i<=Math.sqrt(n);i++){
            if(n%2==0){
                return false;
            }
        }
        return true;
    }
    
    public static void main(String[] args) {
       int n=10;
       for(int i=1;i<=n;i++){
           if(prime(i)){
               System.out.println(i + " ");
           }
       }
    }
}
